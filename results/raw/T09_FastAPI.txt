from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import Optional

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    id: int
    username: str
    permissions: list[str]


class InvoiceCreate(BaseModel):
    title: str
    amount: float
    description: Optional[str] = None
    client_name: str


class Invoice(BaseModel):
    id: int
    title: str
    amount: float
    description: Optional[str] = None
    client_name: str
    owner_id: int


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    # Simulated user lookup based on token
    fake_users = {
        "valid_token_with_permission": User(
            id=1,
            username="authorized_user",
            permissions=["invoices:write", "invoices:read"],
        ),
        "valid_token_without_permission": User(
            id=2,
            username="unauthorized_user",
            permissions=["invoices:read"],
        ),
    }
    user = fake_users.get(token)
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return user


def check_permission(user: User, permission: str) -> bool:
    return permission in user.permissions


invoice_db: list[dict] = []
invoice_id_counter = 1


@app.post("/invoices", response_model=Invoice, status_code=status.HTTP_201_CREATED)
def create_invoice(
    invoice_data: InvoiceCreate,
    current_user: User = Depends(get_current_user),
):
    if not check_permission(current_user, "invoices:write"):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="You do not have permission to create invoices",
        )

    global invoice_id_counter
    new_invoice = Invoice(
        id=invoice_id_counter,
        title=invoice_data.title,
        amount=invoice_data.amount,
        description=invoice_data.description,
        client_name=invoice_data.client_name,
        owner_id=current_user.id,
    )
    invoice_db.append(new_invoice.model_dump())
    invoice_id_counter += 1

    return new_invoice
