import {
  CanActivate,
  Controller,
  ExecutionContext,
  ForbiddenException,
  Get,
  Req,
  UseGuards,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { Request } from 'express';
import { AnalyticsService } from './analytics.service';

interface AuthenticatedUser {
  id: string;
  accountId: string;
  subscription?: {
    plan?: string;
    status?: string;
  };
  account?: {
    featureFlags?: Record<string, boolean>;
  };
}

type AuthenticatedRequest = Request & {
  user: AuthenticatedUser;
};

class PremiumSubscriptionGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest<AuthenticatedRequest>();
    const subscription = request.user?.subscription;

    if (subscription?.plan === 'premium' && subscription?.status === 'active') {
      return true;
    }

    throw new ForbiddenException();
  }
}

class AdvancedAnalyticsFeatureFlagGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest<AuthenticatedRequest>();

    if (request.user?.account?.featureFlags?.['advanced-analytics'] === true) {
      return true;
    }

    throw new ForbiddenException();
  }
}

@Controller('analytics')
export class AnalyticsController {
  constructor(private readonly analyticsService: AnalyticsService) {}

  @Get('advanced')
  @UseGuards(
    AuthGuard('jwt'),
    new PremiumSubscriptionGuard(),
    new AdvancedAnalyticsFeatureFlagGuard(),
  )
  async getAdvancedAnalytics(@Req() request: AuthenticatedRequest) {
    return this.analyticsService.getAdvancedAnalytics(request.user.accountId);
  }
}

// ─── Subscription Guard ───────────────────────────────────────────────────────

@Injectable()
class PremiumSubscriptionGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const user = request.user;

    if (!user) {
      throw new UnauthorizedException('Authentication required');
    }

    const hasActivePremium =
      user.subscription &&
      user.subscription.plan === 'premium' &&
      user.subscription.status === 'active';

    if (!hasActivePremium) {
      throw new ForbiddenException(
        'An active premium subscription is required to access this resource',
      );
    }

    return true;
  }
}

// ─── Feature Flag Guard ───────────────────────────────────────────────────────

@Injectable()
class AdvancedAnalyticsFeatureFlagGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const user = request.user;

    if (!user) {
      throw new UnauthorizedException('Authentication required');
    }

    const featureFlags: string[] = user.featureFlags ?? [];
    const isEnabled = featureFlags.includes('advanced-analytics');

    if (!isEnabled) {
      throw new ForbiddenException(
        'The advanced-analytics feature flag is not enabled for your account',
      );
    }

    return true;
  }
}
