import {
  Controller,
  ExecutionContext,
  ForbiddenException,
  Get,
  Injectable,
  Param,
  UnauthorizedException,
  UseGuards,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { TicketsService } from './tickets.service';

type AuthenticatedUser = {
  role?: string;
  roles?: string[];
};

@Injectable()
class SupportOrAdminGuard extends AuthGuard('jwt') {
  async canActivate(context: ExecutionContext): Promise<boolean> {
    const authenticated = await super.canActivate(context);

    if (!authenticated) {
      throw new UnauthorizedException();
    }

    const request = context
      .switchToHttp()
      .getRequest<{ user?: AuthenticatedUser }>();

    const userRoles = [
      ...(request.user?.role ? [request.user.role] : []),
      ...(Array.isArray(request.user?.roles) ? request.user.roles : []),
    ];

    if (!userRoles.some((role) => role === 'support' || role === 'admin')) {
      throw new ForbiddenException();
    }

    return true;
  }
}

@Controller('tickets')
export class TicketsController {
  constructor(private readonly ticketsService: TicketsService) {}

  @Get(':ticketId/history')
  @UseGuards(new SupportOrAdminGuard())
  async getTicketHistory(@Param('ticketId') ticketId: string): Promise<unknown> {
    return this.ticketsService.findHistory(ticketId);
  }
}

import {
  Controller,
  Get,
  Param,
  UseGuards,
  Request,
  ForbiddenException,
  UnauthorizedException,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { TicketsService } from './tickets.service';

export function Roles(...roles: string[]) {
  return (target: any, key?: string, descriptor?: PropertyDescriptor) => {
    if (descriptor) {
      Reflect.defineMetadata('roles', roles, descriptor.value);
    } else {
      Reflect.defineMetadata('roles', roles, target);
    }
    return descriptor || target;
  };
}

import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.get<string[]>(
      'roles',
      context.getHandler(),
    );

    if (!requiredRoles) {
      return true;
    }

    const request = context.switchToHttp().getRequest();
    const user = request.user;

    if (!user) {
      throw new UnauthorizedException();
    }

    const hasRole = requiredRoles.some((role) =>
      user.roles?.includes(role),
    );

    if (!hasRole) {
      throw new ForbiddenException();
    }

    return true;
  }
}

@Controller('tickets')
@UseGuards(AuthGuard('jwt'))
export class TicketsController {
  constructor(private readonly ticketsService: TicketsService) {}

  @Get(':ticketId/history')
  @UseGuards(RolesGuard)
  @Roles('support', 'admin')
  async getTicketHistory(@Param('ticketId') ticketId: string) {
    return this.ticketsService.findHistory(ticketId);
  }
}
