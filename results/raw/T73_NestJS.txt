import { Request } from 'express';
import { AccountsService } from './accounts.service';
import { Account } from './models/account.model';
import { TransferRequestDto } from './dto/transfer-request.dto';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { AccountOwnerGuard } from './guards/account-owner.guard';
import { PermissionsGuard } from '../auth/guards/permissions.guard';
import { RolesGuard } from '../auth/guards/roles.guard';
import { Permissions } from '../auth/decorators/permissions.decorator';
import { Roles } from '../auth/decorators/roles.decorator';

type AuthenticatedRequest = Request & {
  user: {
    id: string;
  };
};

@Controller('accounts')
@UseGuards(JwtAuthGuard)
export class AccountsController {
  constructor(private readonly accountsService: AccountsService) {}

  @Get()
  async getOwnAccounts(
    @Req() req: AuthenticatedRequest,
  ): Promise<Account[]> {
    return this.accountsService.findAllForUser(req.user.id);
  }

  @Get(':id/transactions')
  @UseGuards(AccountOwnerGuard)
  async getTransactionHistory(
    @Param('id') id: string,
    @Req() req: AuthenticatedRequest,
  ): Promise<unknown[]> {
    return this.accountsService.getTransactionHistory(id, req.user.id);
  }

  @Post(':id/transfer')
  @Permissions('transfers:execute')
  @UseGuards(AccountOwnerGuard, PermissionsGuard)
  async transfer(
    @Param('id') id: string,
    @Body() transferRequestDto: TransferRequestDto,
    @Req() req: AuthenticatedRequest,
  ): Promise<unknown> {
    return this.accountsService.transfer(id, transferRequestDto, req.user.id);
  }

  @Delete(':id')
  @Roles('account_manager')
  @UseGuards(RolesGuard)
  @HttpCode(HttpStatus.NO_CONTENT)
  async closeAccount(@Param('id') id: string): Promise<void> {
    await this.accountsService.closeAccount(id);
  }
}
