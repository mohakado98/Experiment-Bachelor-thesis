from datetime import datetime, timezone
from typing import Annotated, Any

from fastapi import Depends, FastAPI, HTTPException, Security, status

app = FastAPI()

)


SCOPE_IMPLICATIONS = {
    "data:read": {"data:read"},
    "data:internal": {"data:read", "data:internal"},
    "data:admin": {"data:read", "data:internal", "data:admin"},
}


class User(BaseModel):

async def get_current_user_with_scopes(
    security_scopes: SecurityScopes,
    token: Annotated[str, Depends(oauth2_scheme)],
) -> User:
    token_scopes = set(token.replace(",", " ").split())

    if not token_scopes:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )

    effective_scopes: set[str] = set()
    for scope in token_scopes:
        effective_scopes.update(SCOPE_IMPLICATIONS.get(scope, set()))

    required_scopes = set(security_scopes.scopes)

    if not required_scopes.issubset(effective_scopes):
        authenticate_value = f'Bearer scope="{security_scopes.scope_str}"'
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Insufficient scope",
            headers={"WWW-Authenticate": authenticate_value},
        )

    return User(username="authenticated-user", scopes=sorted(token_scopes))

    "data:read": 1,
    "data:internal": 2,
    "data:admin": 3,

def get_current_user_with_scopes(security_scopes: SecurityScopes, token: str = Depends(oauth2_scheme)) -> User:
    if security_scopes.scopes:
        authenticate_value = f'Bearer scope="{security_scopes.scope_str}"'
    else:
        authenticate_value = "Bearer"

    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": authenticate_value},
    )

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise credentials_exception
        token_scopes = payload.get("scopes", [])
        token_data = TokenData(scopes=token_scopes, username=username)
    except PyJWTError:
        raise credentials_exception

    user = User(username=token_data.username, scopes=token_data.scopes)

    for required_scope in security_scopes.scopes:
        required_level = SCOPE_HIERARCHY.get(required_scope, 0)
        user_max_level = max(
            (SCOPE_HIERARCHY.get(s, 0) for s in user.scopes),
            default=0,
        )
        if user_max_level < required_level:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Insufficient scope. Required: '{required_scope}'. "
                       f"Your scopes: {user.scopes}",
                headers={"WWW-Authenticate": authenticate_value},
            )

    return user


def user_has_minimum_scope(user: User, minimum_scope: str) -> bool:
    required_level = SCOPE_HIERARCHY.get(minimum_scope, 0)
    user_max_level = max(
        (SCOPE_HIERARCHY.get(s, 0) for s in user.scopes),
        default=0,
    )
    return user_max_level >= required_level
