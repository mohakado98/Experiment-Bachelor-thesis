from typing import Any
from uuid import uuid4
import inspect

from fastapi import APIRouter, Depends, FastAPI, HTTPException, Response, status

from auth import get_current_user, require_admin
from models import Course


app = FastAPI()

users_db: list[Any] = []
courses_db: dict[str, Course] = {}


async def require_admin_user(current_user: Any = Depends(get_current_user)) -> Any:
    result = require_admin(current_user)
    if inspect.isawaitable(result):
        result = await result
    return result if result is not None else current_user


admin_router = APIRouter(
    prefix="/admin",
    tags=["admin"],
    dependencies=[Depends(require_admin_user)],
)


@admin_router.get("/users")
async def get_all_users() -> list[Any]:
    return users_db


@admin_router.post("/courses", status_code=status.HTTP_201_CREATED)
async def create_course(course: Course) -> dict[str, Any]:
    course_id = str(uuid4())
    courses_db[course_id] = course
    return {"id": course_id, "course": course}


@admin_router.put("/courses/{course_id}")
async def update_course(course_id: str, course: Course) -> dict[str, Any]:
    if course_id not in courses_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Course not found",
        )

    courses_db[course_id] = course
    return {"id": course_id, "course": course}


@admin_router.delete("/courses/{course_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_course(course_id: str) -> Response:
    if course_id not in courses_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Course not found",
        )

    del courses_db[course_id]
    return Response(status_code=status.HTTP_204_NO_CONTENT)


app.include_router(admin_router)
