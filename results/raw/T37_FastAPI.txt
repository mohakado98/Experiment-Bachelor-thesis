from inspect import isawaitable
from typing import Any

app = FastAPI()


async def _maybe_await(value: Any) -> Any:
    if isawaitable(value):
        return await value
    return value


async def enforce_patient_record_access(patient_id: int, user: Any) -> None:
    try:
        role_allowed = await _maybe_await(require_clinician_role(user))
        if role_allowed is False:
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Forbidden")

        has_relationship = await _maybe_await(check_care_relationship(user, patient_id))
        if not has_relationship:
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Forbidden")

            raise
