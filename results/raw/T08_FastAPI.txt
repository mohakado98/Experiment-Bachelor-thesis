from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from typing import Optional

from models import UserProfile, ProfileUpdate
from auth import get_current_user

router = APIRouter(prefix="/account", tags=["account"])

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


@router.get("/profile", response_model=UserProfile, status_code=status.HTTP_200_OK)
async def get_profile(current_user: UserProfile = Depends(get_current_user)):
    if current_user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return current_user


@router.put("/profile", response_model=UserProfile, status_code=status.HTTP_200_OK)
async def update_profile(
    profile_update: ProfileUpdate,
    current_user: UserProfile = Depends(get_current_user),
):
    if current_user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    updated_data = profile_update.dict(exclude_unset=True)
    updated_user = current_user.copy(update=updated_data)
    return updated_user


@router.delete("", status_code=status.HTTP_204_NO_CONTENT)
async def delete_account(current_user: UserProfile = Depends(get_current_user)):
    if current_user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return None


from fastapi import FastAPI

app = FastAPI()
app.include_router(router)
