import { isObservable, lastValueFrom } from 'rxjs';
import { ResourceOwnerGuard } from './guards/resource-owner.guard';
import { DelegatedAccessGuard } from './guards/delegated-access.guard';
import { AdminAuditGuard } from './guards/admin-audit.guard';
import { SharedResourcesService } from './shared-resources.service';

type AuthenticatedUser = {
  id: string;
  isAdmin?: boolean;
  admin?: boolean;
  role?: string;
  roles?: Array<string | { name?: string }>;
  [key: string]: unknown;
};

type AuthenticatedRequest = Request & {
  user?: AuthenticatedUser;
};

function SharedResourceAccessGuard(): Type<CanActivate> {
  @Injectable()
  class SharedResourceAccessGuardMixin implements CanActivate {
    constructor(
      private readonly resourceOwnerGuard: ResourceOwnerGuard,
      private readonly delegatedAccessGuard: DelegatedAccessGuard,
      private readonly adminAuditGuard: AdminAuditGuard,
    ) {}

    async canActivate(context: ExecutionContext): Promise<boolean> {
      const request = context.switchToHttp().getRequest<AuthenticatedRequest>();

      if (!request.user) {
        throw new UnauthorizedException();
      }

      if (await this.evaluateGuard(this.resourceOwnerGuard, context)) {
        return true;
      }

      if (this.isAdmin(request.user)) {
        if (
          await this.evaluateGuard(this.adminAuditGuard, context, {
            rethrowForbidden: true,
          })
        ) {
          return true;
        }

        throw new ForbiddenException('Admin access requires a logged justification');
      }

      if (await this.evaluateGuard(this.delegatedAccessGuard, context)) {
        return true;
      }

      throw new ForbiddenException();
    }

    private async evaluateGuard(
      guard: CanActivate,
      context: ExecutionContext,
      options: { rethrowForbidden?: boolean } = {},
    ): Promise<boolean> {
      try {
        const result = guard.canActivate(context);
        const resolved = isObservable(result) ? await lastValueFrom(result) : await result;
        return resolved === true;
      } catch (error) {
        if (this.isHttpStatus(error, 401)) {
          throw error;
        }

        if (options.rethrowForbidden && this.isHttpStatus(error, 403)) {
          throw error;
        }

        return false;
      }
    }

    private isHttpStatus(error: unknown, status: number): boolean {
      return error instanceof HttpException && error.getStatus() === status;
    }

    private isAdmin(user: AuthenticatedUser): boolean {
      if (user.isAdmin === true || user.admin === true) {
        return true;
      }

      const roles = [
        user.role,
        ...(Array.isArray(user.roles) ? user.roles : []),
      ]
        .map((role) => {
          if (typeof role === 'string') {
            return role;
          }

          if (role && typeof role.name === 'string') {
            return role.name;
          }

          return '';
        })
        .map((role) => role.toLowerCase());

      return roles.includes('admin') || roles.includes('administrator');
    }
  }

  return mixin(SharedResourceAccessGuardMixin);
}

@Controller('shared')
export class SharedResourcesController {
  constructor(private readonly sharedResourcesService: SharedResourcesService) {}

  @Get(':resourceId/access')
  @UseGuards(AuthGuard('jwt'), SharedResourceAccessGuard())
  async getSharedResource(
    @Param('resourceId') resourceId: string,
    @Req() request: AuthenticatedRequest,
  ): Promise<unknown> {
    return this.sharedResourcesService.getSharedResource(resourceId, request.user!);
  }
}

// guards/any-access.guard.ts
import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
  mixin,
  Type,
} from '@nestjs/common';
import { ModuleRef } from '@nestjs/core';

export function AnyAccessGuard(...guardTypes: Type<CanActivate>[]): Type<CanActivate> {
  @Injectable()
  class MixedAccessGuard implements CanActivate {
    private guards: CanActivate[];

    constructor(private readonly moduleRef: ModuleRef) {
      this.guards = guardTypes.map((GuardType) =>
        this.moduleRef.get(GuardType, { strict: false }),
      );
    }

    async canActivate(context: ExecutionContext): Promise<boolean> {
      const results = await Promise.allSettled(
        this.guards.map((guard) =>
          Promise.resolve(guard.canActivate(context)),
        ),
      );

      const anyGranted = results.some(
        (result) => result.status === 'fulfilled' && result.value === true,
      );

      if (!anyGranted) {
        throw new ForbiddenException(
          'Access denied: insufficient permissions or missing audit justification',
        );
      }

      return true;
    }
  }

  return mixin(MixedAccessGuard);
}

import { SharedResourcesService } from '../shared-resources.service';

export class ResourceOwnerGuard implements CanActivate {
  constructor(private readonly sharedResourcesService: SharedResourcesService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const resourceId = request.params.resourceId;

    const resource = await this.sharedResourcesService.findById(resourceId);

    if (!resource) {
      throw new ForbiddenException('Resource not found');
    }

    if (resource.ownerId !== user.id) {
      throw new ForbiddenException('User is not the resource owner');
    }

    return true;
  }
}

import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
} from '@nestjs/common';
import { SharedResourcesService } from '../shared-resources.service';

@Injectable()
export class DelegatedAccessGuard implements CanActivate {
  constructor(private readonly sharedResourcesService: SharedResourcesService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const resourceId = request.params.resourceId;
    const shareToken =
      request.headers['x-share-token'] || request.query.shareToken;

    if (!shareToken) {
      throw new ForbiddenException('No share token provided');
    }

    const isGranted = await this.sharedResourcesService.validateShareToken(
      resourceId,
      user.id,
      shareToken as string,
    );

    if (!isGranted) {
      throw new ForbiddenException(
        'Invalid or expired share token for this user',
      );
    }

    return true;
  }
}

// guards/admin-audit.guard.ts
import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
} from '@nestjs/common';
import { SharedResourcesService } from '../shared-resources.service';

@Injectable()
export class AdminAuditGuard implements CanActivate {
  constructor(private readonly sharedResourcesService: SharedResourcesService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const resourceId = request.params.resourceId;
    const justification = request.headers['x-admin-justification'];

    if (!user.roles || !user.roles.includes('admin')) {
      throw new ForbiddenException('User is not an administrator');
    }

    if (!justification || (justification as string).trim().length === 0) {
      throw new ForbiddenException(
        'Administrators must provide a justification via x-admin-justification header to access resources they do not own',
      );
    }

    await this.sharedResourcesService.logAdminAccess({
      adminId: user.id,
      resourceId,
      justification: justification as string,
      accessedAt: new Date(),
    });

    return true;
  }
}

// shared-resources.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';

export interface SharedResource {
  id: string;
  ownerId: string;
  data: unknown;
  createdAt: Date;
}

export interface AdminAccessLog {
  adminId: string;
  resourceId: string;
  justification: string;
  accessedAt: Date;
}

@Injectable()
export class SharedResourcesService {
  private readonly resources: Map<string, SharedResource> = new Map();
  private readonly shareTokens: Map<string, { userId: string; token: string; expiresAt: Date }[]> = new Map();
  private readonly adminAccessLogs: AdminAccessLog[] = [];

  async findById(resourceId: string): Promise<SharedResource> {
    const resource = this.resources.get(resourceId);
    if (!resource) {
      throw new NotFoundException(`Resource ${resourceId} not found`);
    }
    return resource;
  }

  async validateShareToken(
    resourceId: string,
    userId: string,
    token: string,
  ): Promise<boolean> {
    const tokens = this.shareTokens.get(resourceId) || [];
    const now = new Date();
    return tokens.some(
      (entry) =>
        entry.userId === userId &&
        entry.token === token &&
        entry.expiresAt > now,
    );
  }

  async logAdminAccess(log: AdminAccessLog): Promise<void> {
    this.adminAccessLogs.push(log);
  }
}
