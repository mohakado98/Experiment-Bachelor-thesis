import inspect

from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel
from datetime import date
from decimal import Decimal


router = APIRouter()


class SalaryHistoryEntry(BaseModel):
    employee_id: int
    salary: Decimal
    currency: str
    effective_date: date
    end_date: date | None = None


SALARY_HISTORY: dict[int, list[SalaryHistoryEntry]] = {}


@router.get(
    "/employees/{employee_id}/salary-history",
    response_model=list[SalaryHistoryEntry],
    status_code=status.HTTP_200_OK,
)
async def get_employee_salary_history(
    employee_id: int,
    current_user: Annotated[Any, Depends(get_current_user)],
) -> list[SalaryHistoryEntry]:
    authorized = require_hr_or_self(current_user, employee_id)

    if inspect.isawaitable(authorized):
        authorized = await authorized

    if authorized is False:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Not authorized to view this employee's salary history",
        )

    return SALARY_HISTORY.get(employee_id, [])

def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id: int = payload.get("user_id")
        if user_id is None:
            raise credentials_exception
        user = FAKE_USERS_DB.get(user_id)
        if user is None:
            raise credentials_exception
        return user
    except jwt.PyJWTError:
        raise credentials_exception

def require_hr_or_self(current_user: User, employee_id: int) -> None:
    if current_user.role == "hr_manager":
        return
    if current_user.employee_id == employee_id:
        return
    raise HTTPException(
        status_code=status.HTTP_403_FORBIDDEN,
        detail="You do not have permission to access this resource",
    )

@app.get("/employees/{employee_id}/salary-history", response_model=SalaryHistoryResponse)
def get_salary_history(
    employee_id: int,
    current_user: User = Depends(get_current_user),
):
    require_hr_or_self(current_user, employee_id)

    salary_records = FAKE_SALARY_DB.get(employee_id)
    if salary_records is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Employee with id {employee_id} not found",
        )

    return SalaryHistoryResponse(
        employee_id=employee_id,
        salary_history=salary_records,
    )
