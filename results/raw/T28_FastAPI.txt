from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import Optional
import random

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    id: int
    username: str
    email: str
    account_id: str
    is_active: bool = True


class AISuggestion(BaseModel):
    suggestion_id: str
    content: str
    confidence: float
    category: str


class AISuggestionsResponse(BaseModel):
    user_id: int
    username: str
    feature: str
    suggestions: list[AISuggestion]
    message: str


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )

    fake_users_db = {
        "valid_token_123": User(
            id=1,
            username="john_doe",
            email="john@example.com",
            account_id="acc_001",
        ),
        "valid_token_456": User(
            id=2,
            username="jane_smith",
            email="jane@example.com",
            account_id="acc_002",
        ),
    }

    user = fake_users_db.get(token)
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid authentication credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )

    return user


def check_feature_flag(user: User, feature: str) -> bool:
    feature_flags_db = {
        "acc_001": ["ai_assistant", "dark_mode", "beta_dashboard"],
        "acc_002": ["dark_mode"],
        "acc_003": ["ai_assistant"],
    }

    account_flags = feature_flags_db.get(user.account_id, [])
    return feature in account_flags


def generate_ai_suggestions(user: User) -> list[AISuggestion]:
    suggestions = [
        AISuggestion(
            suggestion_id=f"sug_{random.randint(1000, 9999)}",
            content="Consider optimizing your workflow by automating repetitive tasks using our new batch processing feature.",
            confidence=0.92,
            category="productivity",
        ),
        AISuggestion(
            suggestion_id=f"sug_{random.randint(1000, 9999)}",
            content="Based on your usage patterns, switching to dark mode could reduce eye strain during long sessions.",
            confidence=0.87,
            category="user_experience",
        ),
        AISuggestion(
            suggestion_id=f"sug_{random.randint(1000, 9999)}",
            content="Your account shows high API usage. Upgrading your plan could unlock higher rate limits and better performance.",
            confidence=0.78,
            category="account_management",
        ),
    ]
    return suggestions


@app.get(
    "/features/beta/ai-assistant",
    response_model=AISuggestionsResponse,
    status_code=status.HTTP_200_OK,
    summary="Get AI-generated suggestions",
    description="Returns AI-generated suggestions for authenticated users with the ai_assistant feature flag enabled.",
    responses={
        200: {"description": "Successfully retrieved AI suggestions"},
        401: {"description": "User is not authenticated"},
        403: {"description": "User does not have access to the AI assistant feature"},
    },
)
async def get_ai_assistant_suggestions(
    current_user: User = Depends(get_current_user),
) -> AISuggestionsResponse:
    feature_name = "ai_assistant"

    if not check_feature_flag(current_user, feature_name):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=f"Access denied. The '{feature_name}' feature is not enabled for your account. "
                   f"Please contact support to request access.",
        )

    ai_suggestions = generate_ai_suggestions(current_user)

    return AISuggestionsResponse(
        user_id=current_user.id,
        username=current_user.username,
        feature=feature_name,
        suggestions=ai_suggestions,
        message="AI suggestions generated successfully.",
    )


if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
