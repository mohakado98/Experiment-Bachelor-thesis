from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import Optional
import uuid

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    id: str
    email: str
    is_verified: bool


class ListingCreate(BaseModel):
    title: str
    description: str
    price: float
    category: Optional[str] = None


class ListingResponse(BaseModel):
    id: str
    title: str
    description: str
    price: float
    category: Optional[str]
    seller_id: str
    seller_email: str


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    fake_users = {
        "verified_token": User(id="1", email="verified@example.com", is_verified=True),
        "unverified_token": User(id="2", email="unverified@example.com", is_verified=False),
    }
    user = fake_users.get(token)
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return user


def require_verified_email(current_user: User = Depends(get_current_user)) -> User:
    if not current_user.is_verified:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Email address has not been verified. Please verify your email before creating listings.",
        )
    return current_user


@app.post("/listings", response_model=ListingResponse, status_code=status.HTTP_201_CREATED)
async def create_listing(
    listing: ListingCreate,
    current_user: User = Depends(require_verified_email),
):
    new_listing = ListingResponse(
        id=str(uuid.uuid4()),
        title=listing.title,
        description=listing.description,
        price=listing.price,
        category=listing.category,
        seller_id=current_user.id,
        seller_email=current_user.email,
    )
    return new_listing
