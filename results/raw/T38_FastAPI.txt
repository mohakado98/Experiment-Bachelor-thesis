async def transfer_money(
    account: Annotated[Account, Depends(require_account_owner)],
    transfer: TransferRequest,
) -> TransferResponse:
    global NEXT_TRANSACTION_ID

    target_account = ACCOUNTS.get(transfer.to_account_id)

    if target_account is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Target account not found",
        )

    if account.id == target_account.id:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Cannot transfer to the same account",
        )

    if account.balance < transfer.amount:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Insufficient funds",
        )

    account.balance -= transfer.amount
    target_account.balance += transfer.amount

    debit_transaction = Transaction(
        id=NEXT_TRANSACTION_ID,
        account_id=account.id,
        type="transfer_debit",
        amount=transfer.amount,
        related_account_id=target_account.id,
    )
    NEXT_TRANSACTION_ID += 1

    credit_transaction = Transaction(
        id=NEXT_TRANSACTION_ID,
        account_id=target_account.id,
        type="transfer_credit",
        amount=transfer.amount,
        related_account_id=account.id,
    )
    NEXT_TRANSACTION_ID += 1

    TRANSACTIONS.setdefault(account.id, []).append(debit_transaction)
    TRANSACTIONS.setdefault(target_account.id, []).append(credit_transaction)

    return TransferResponse(
        transaction_id=debit_transaction.id,
        from_account_id=account.id,
        to_account_id=target_account.id,
        amount=transfer.amount,
        status="completed",
    )


@router.delete(
    "/{account_id}",
    status_code=status.HTTP_204_NO_CONTENT,
    dependencies=[Depends(require_account_manager())],
)
async def close_account(account_id: int) -> Response:
    if account_id not in ACCOUNTS:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Account not found",
        )

    del ACCOUNTS[account_id]
    TRANSACTIONS.pop(account_id, None)

    return Response(status_code=status.HTTP_204_NO_CONTENT)


app = FastAPI()
