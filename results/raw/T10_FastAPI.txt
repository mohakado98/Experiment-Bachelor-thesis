from fastapi import APIRouter, Depends, HTTPException, Response, status

router = APIRouter(prefix="/products", tags=["products"])

next_product_id = 1


def product_with_id(product: Product, product_id: int) -> Product:
    if hasattr(product, "model_copy"):
        return product.model_copy(update={"id": product_id})
    return product.copy(update={"id": product_id})


@router.get("", response_model=list[Product])
async def get_products() -> list[Product]:
    return list(products_db.values())

            detail="Product not found",
        )

    "",
    response_model=Product,
    status_code=status.HTTP_201_CREATED,
    dependencies=[Depends(get_current_user)],
)
async def create_product(product: Product) -> Product:
    global next_product_id

    product_id = getattr(product, "id", None)
    if product_id is None:
        product_id = next_product_id
        next_product_id += 1
    elif product_id in products_db:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Product already exists",
        )
    else:
        next_product_id = max(next_product_id, product_id + 1)

    product = product_with_id(product, product_id)
    products_db[product_id] = product
    return product

    "/{product_id}",
    response_model=Product,
    dependencies=[Depends(get_current_user)],
)

        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Product not found",
        )

    product = product_with_id(product, product_id)
    products_db[product_id] = product
    return product

    status_code=status.HTTP_204_NO_CONTENT,
    dependencies=[Depends(require_admin)],

            status_code=status.HTTP_404_NOT_FOUND,
