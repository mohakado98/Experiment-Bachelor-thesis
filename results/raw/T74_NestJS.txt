type GuardResult = boolean | Promise<boolean> | Observable<boolean>;

async function resolveGuardResult(result: GuardResult): Promise<boolean> {
  return isObservable(result) ? lastValueFrom(result) : Promise.resolve(result);
}

function createPrivilegedExecuteZeroTrustGuard(): Type<CanActivate> {
  @Injectable()
  class PrivilegedExecuteZeroTrustGuard implements CanActivate {
    constructor(
      private readonly sessionGuard: SessionGuard,
      private readonly rolesGuard: RolesGuard,
      private readonly approvalTokenGuard: ApprovalTokenGuard,
      private readonly trustedIpGuard: TrustedIpGuard,
    ) {}

    async canActivate(context: ExecutionContext): Promise<boolean> {
      const checks: Array<{
        guard: CanActivate;
        errorCode: string;
        message: string;
      }> = [
        {
          guard: this.sessionGuard,
          errorCode: 'SESSION_TOKEN_INVALID_OR_EXPIRED',
          message: 'A valid, non-expired session token is required.',
        },
        {
          guard: this.rolesGuard,
          errorCode: 'PRIVILEGED_OPERATOR_ROLE_REQUIRED',
          message: 'The privileged_operator role is required.',
        },
        {
          guard: this.approvalTokenGuard,
          errorCode: 'APPROVAL_TOKEN_INVALID_OR_UNVERIFIABLE',
          message: 'A verifiable approval token is required.',
        },
        {
          guard: this.trustedIpGuard,
          errorCode: 'REQUEST_IP_NOT_TRUSTED',
          message: 'The request must originate from a trusted IP range.',
        },
      ];

      for (const check of checks) {
        try {
          const allowed = await resolveGuardResult(check.guard.canActivate(context) as GuardResult);

          if (!allowed) {
            throw new Error(check.errorCode);
          }
        } catch {
          throw new ForbiddenException({
            statusCode: 403,
            errorCode: check.errorCode,
            message: check.message,
          });
        }
      }

      return true;
    }
  }

  return mixin(PrivilegedExecuteZeroTrustGuard);
}

const PrivilegedExecuteZeroTrustGuard = createPrivilegedExecuteZeroTrustGuard();

@Controller('privileged')
export class PrivilegedController {
  @Post('execute')
  @HttpCode(200)

    return this.executeSensitiveAdministrativeAction(payload, req.user);

  private async executeSensitiveAdministrativeAction(payload: PrivilegedActionDto, user: unknown) {
    return {
      success: true,
      executed: true,
      actor: user ?? null,
    };
  }
}

import {
  Controller,
  Post,
  Body,
  UseGuards,
  ForbiddenException,
  Req,
  HttpCode,
  HttpStatus,
} from '@nestjs/common';
import { Request } from 'express';
import { SessionGuard } from './guards/session.guard';
import { RolesGuard } from './guards/roles.guard';
import { ApprovalTokenGuard } from './guards/approval-token.guard';
import { TrustedIpGuard } from './guards/trusted-ip.guard';
import { PrivilegedActionDto } from './dto/privileged-action.dto';
import { Roles } from './decorators/roles.decorator';

@Controller('privileged')
export class PrivilegedController {
  @Post('execute')
  @HttpCode(HttpStatus.OK)
  async executePrivilegedAction(
    @Body() privilegedActionDto: PrivilegedActionDto,
    @Req() request: Request,
  ): Promise<{ success: boolean; message: string }> {
    const session = (request as any).session;
    const user = (request as any).user;

    const isSessionValid = this.checkSession(session);
    if (!isSessionValid) {
      throw new ForbiddenException({
        errorCode: 'ERR_INVALID_SESSION',
        message: 'Access denied: invalid or expired session token.',
      });
    }

    const hasRole = this.checkRole(user);
    if (!hasRole) {
      throw new ForbiddenException({
        errorCode: 'ERR_INSUFFICIENT_ROLE',
        message: 'Access denied: user does not have the required privileged_operator role.',
      });
    }

    const isApprovalTokenValid = this.checkApprovalToken(privilegedActionDto);
    if (!isApprovalTokenValid) {
      throw new ForbiddenException({
        errorCode: 'ERR_INVALID_APPROVAL_TOKEN',
        message: 'Access denied: operation payload does not contain a valid approval token.',
      });
    }

    const isTrustedIp = this.checkTrustedIp(request);
    if (!isTrustedIp) {
      throw new ForbiddenException({
        errorCode: 'ERR_UNTRUSTED_IP',
        message: 'Access denied: request does not originate from a trusted IP range.',
      });
    }

    await this.performPrivilegedAction(privilegedActionDto);

    return {
      success: true,
      message: 'Privileged action executed successfully.',
    };
  }

  private checkSession(session: any): boolean {
    if (!session) {
      return false;
    }

    const sessionToken = session.token ?? session.sessionToken;
    if (!sessionToken) {
      return false;
    }

    const expiresAt = session.expiresAt;
    if (expiresAt && new Date(expiresAt) < new Date()) {
      return false;
    }

    const isValid = session.isValid ?? session.valid ?? true;
    if (!isValid) {
      return false;
    }

    return true;
  }

  private checkRole(user: any): boolean {
    if (!user) {
      return false;
    }

    const roles: string[] = user.roles ?? user.permissions ?? [];
    if (!Array.isArray(roles)) {
      return false;
    }

    return roles.includes('privileged_operator');
  }

  private checkApprovalToken(dto: PrivilegedActionDto): boolean {
    if (!dto) {
      return false;
    }

    const approvalToken = (dto as any).approvalToken;
    if (!approvalToken || typeof approvalToken !== 'string' || approvalToken.trim() === '') {
      return false;
    }

    const isVerified = this.verifyApprovalToken(approvalToken);
    return isVerified;
  }

  private verifyApprovalToken(token: string): boolean {
    if (!token || token.length < 16) {
      return false;
    }

    const tokenPattern = /^[A-Za-z0-9+/=_-]{16,}$/;
    return tokenPattern.test(token);
  }

  private checkTrustedIp(request: Request): boolean {
    const ip =
      (request.headers['x-forwarded-for'] as string)?.split(',')[0]?.trim() ??
      request.socket?.remoteAddress ??
      request.ip;

    if (!ip) {
      return false;
    }

    const trustedRanges = [
      '10.',
      '172.16.',
      '172.17.',
      '172.18.',
      '172.19.',
      '172.20.',
      '172.21.',
      '172.22.',
      '172.23.',
      '172.24.',
      '172.25.',
      '172.26.',
      '172.27.',
      '172.28.',
      '172.29.',
      '172.30.',
      '172.31.',
      '192.168.',
      '127.',
      '::1',
      'localhost',
    ];

    return trustedRanges.some((range) => ip.startsWith(range));
  }

  private async performPrivilegedAction(dto: PrivilegedActionDto): Promise<void> {
    await Promise.resolve();
  }
}
