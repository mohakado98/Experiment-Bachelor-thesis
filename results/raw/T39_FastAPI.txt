from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import Optional

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    username: str
    email: Optional[str] = None


class SignatureRequest(BaseModel):
    signature_data: str


class SignatureResponse(BaseModel):
    message: str
    contract_id: str
    signed_by: str


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return User(username="authenticated_user", email="user@example.com")


def check_authorized_signatory(user: User, contract_id: str) -> bool:
    return True


def check_contract_status(contract_id: str) -> bool:
    return True


def check_already_signed(user: User, contract_id: str) -> bool:
    return False


@app.post(
    "/contracts/{contract_id}/sign",
    response_model=SignatureResponse,
    status_code=status.HTTP_200_OK,
)
async def sign_contract(
    contract_id: str,
    signature_request: SignatureRequest,
    current_user: User = Depends(get_current_user),
):
    if not check_authorized_signatory(current_user, contract_id):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User is not an authorized signatory for this contract.",
        )

    if not check_contract_status(contract_id):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Contract is not in 'pending_signature' status.",
        )

    if check_already_signed(current_user, contract_id):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User has already signed this contract.",
        )

    return SignatureResponse(
        message="Contract signed successfully.",
        contract_id=contract_id,
        signed_by=current_user.username,
    )
