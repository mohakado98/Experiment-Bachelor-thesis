import {
  Controller,
  Get,
  Param,
  UseGuards,
  Request,
  HttpCode,
  HttpStatus,
  ForbiddenException,
  UnauthorizedException,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { RolesGuard } from './guards/roles.guard';
import { CareRelationshipGuard } from './guards/care-relationship.guard';
import { PrivacyRestrictionGuard } from './guards/privacy-restriction.guard';
import { Roles } from './decorators/roles.decorator';
import { MedicalRecordsService } from './medical-records.service';

@Controller('patients')
export class PatientsController {
  constructor(private readonly medicalRecordsService: MedicalRecordsService) {}

  @Get(':patientId/records')
  @UseGuards(
    AuthGuard('jwt'),
    RolesGuard,
    CareRelationshipGuard,
    PrivacyRestrictionGuard,
  )
  @Roles('clinician', 'doctor')
  @HttpCode(HttpStatus.OK)
  async getPatientRecords(
    @Param('patientId') patientId: string,
    @Request() req: any,
  ) {
    if (!req.user) {
      throw new UnauthorizedException();
    }

    const records = await this.medicalRecordsService.findByPatientId(patientId);

    if (!records) {
      throw new ForbiddenException();
    }

    return records;
  }
}

// guards/roles.guard.ts
import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<string[]>('roles', [
      context.getHandler(),
      context.getClass(),
    ]);

    if (!requiredRoles) {
      return true;
    }

    const { user } = context.switchToHttp().getRequest();

    if (!user) {
      return false;
    }

    const hasRole = requiredRoles.some((role) =>
      user.roles?.includes(role),
    );

    if (!hasRole) {
      throw new ForbiddenException(
        'You do not have the required role to access this resource.',
      );
    }

    return true;
  }
}

// guards/care-relationship.guard.ts
import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
} from '@nestjs/common';
import { CliniciansService } from '../clinicians.service';

@Injectable()
export class CareRelationshipGuard implements CanActivate {
  constructor(private readonly cliniciansService: CliniciansService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const patientId = request.params.patientId;

    if (!user || !patientId) {
      throw new ForbiddenException(
        'Missing user or patient information.',
      );
    }

    const hasActiveRelationship =
      await this.cliniciansService.hasActiveCareRelationship(
        user.id,
        patientId,
      );

    if (!hasActiveRelationship) {
      throw new ForbiddenException(
        'You are not assigned to this patient.',
      );
    }

    return true;
  }
}

// guards/privacy-restriction.guard.ts
import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
} from '@nestjs/common';
import { CliniciansService } from '../clinicians.service';
import { MedicalRecordsService } from '../medical-records.service';

@Injectable()
export class PrivacyRestrictionGuard implements CanActivate {
  constructor(
    private readonly cliniciansService: CliniciansService,
    private readonly medicalRecordsService: MedicalRecordsService,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const patientId = request.params.patientId;

    if (!user || !patientId) {
      throw new ForbiddenException(
        'Missing user or patient information.',
      );
    }

    const clinicianDepartment = await this.cliniciansService.getDepartment(
      user.id,
    );

    const isRestricted =
      await this.medicalRecordsService.isBlockedForDepartment(
        patientId,
        clinicianDepartment,
      );

    if (isRestricted) {
      throw new ForbiddenException(
        'Access to this patient record is restricted for your department.',
      );
    }

    return true;
  }
}
