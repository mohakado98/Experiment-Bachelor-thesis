import { Body, Controller, Param, Post, Req, UseGuards } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { Request } from 'express';
import { AuthorizedSignatoryGuard } from './guards/authorized-signatory.guard';
import { ContractStatusGuard } from './guards/contract-status.guard';
import { DuplicateSignatureGuard } from './guards/duplicate-signature.guard';
import { ContractsService } from './contracts.service';

class SignContractDto {
  signature: string;
}

interface AuthenticatedRequest extends Request {
  user: {
    id: string;
  };
}

@Controller('contracts')
export class ContractsController {
  constructor(private readonly contractsService: ContractsService) {}

  @Post(':contractId/sign')
  @UseGuards(
    AuthGuard('jwt'),
    AuthorizedSignatoryGuard,
    ContractStatusGuard,
    DuplicateSignatureGuard,
  )
  async signContract(
    @Param('contractId') contractId: string,
    @Body() signContractDto: SignContractDto,
    @Req() req: AuthenticatedRequest,
  ) {
    return this.contractsService.signContract({
      contractId,
      userId: req.user.id,
      signature: signContractDto.signature,
    });
  }
}

import {
  Controller,
  Post,
  Param,
  Req,
  UseGuards,
  HttpCode,
  HttpStatus,
  ForbiddenException,
} from '@nestjs/common';
import { Request } from 'express';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { AuthorizedSignatoryGuard } from '../guards/authorized-signatory.guard';
import { ContractStatusGuard } from '../guards/contract-status.guard';
import { DuplicateSignatureGuard } from '../guards/duplicate-signature.guard';
import { ContractsService } from './contracts.service';

@Controller('contracts')
export class ContractsController {
  constructor(private readonly contractsService: ContractsService) {}

  @Post(':contractId/sign')
  @HttpCode(HttpStatus.OK)
  @UseGuards(
    JwtAuthGuard,
    AuthorizedSignatoryGuard,
    ContractStatusGuard,
    DuplicateSignatureGuard,
  )
  async signContract(
    @Param('contractId') contractId: string,
    @Req() req: Request & { user: { id: string; [key: string]: any } },
  ) {
    const userId = req.user.id;

    const result = await this.contractsService.recordSignature(
      contractId,
      userId,
    );

    if (!result) {
      throw new ForbiddenException(
        'Unable to record signature for this contract.',
      );
    }

    return {
      message: 'Contract signed successfully.',
      contractId,
      signedBy: userId,
      signedAt: new Date().toISOString(),
    };
  }
}
