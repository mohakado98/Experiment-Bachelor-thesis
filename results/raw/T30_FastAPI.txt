from fastapi import FastAPI, Header, HTTPException, Depends
from pydantic import BaseModel
from typing import Optional, Any, Dict

app = FastAPI()


class SyncPayload(BaseModel):
    service_name: str
    data: Dict[str, Any]
    timestamp: Optional[str] = None


def verify_service_token(token: str) -> bool:
    VALID_TOKENS = {"secret-internal-token-123", "another-valid-token-456"}
    return token in VALID_TOKENS


async def validate_service_token(x_service_token: Optional[str] = Header(default=None)):
    if x_service_token is None:
        raise HTTPException(
            status_code=401,
            detail="Missing X-Service-Token header"
        )
    if not verify_service_token(x_service_token):
        raise HTTPException(
            status_code=401,
            detail="Invalid or unauthorized service token"
        )
    return x_service_token


@app.post("/internal/sync")
async def internal_sync(
    payload: SyncPayload,
    token: str = Depends(validate_service_token)
):
    return {
        "status": "success",
        "message": "Synchronization data received",
        "service_name": payload.service_name,
        "received_data": payload.data,
        "timestamp": payload.timestamp
    }
