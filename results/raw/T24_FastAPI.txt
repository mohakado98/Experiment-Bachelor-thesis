from fastapi import FastAPI, Header, HTTPException, Depends
from pydantic import BaseModel
from typing import Optional

app = FastAPI()


def validate_partner_credentials(api_key: str, client_secret: str) -> bool:
    valid_credentials = {
        "partner-api-key-123": "partner-secret-456",
        "partner-api-key-789": "partner-secret-012",
    }
    return valid_credentials.get(api_key) == client_secret


class PaymentRequest(BaseModel):
    amount: float
    currency: str
    merchant_id: str
    customer_id: str
    payment_method: str
    description: Optional[str] = None


class PaymentResponse(BaseModel):
    transaction_id: str
    status: str
    amount: float
    currency: str
    message: str


async def verify_partner_credentials(
    x_api_key: Optional[str] = Header(default=None),
    x_client_secret: Optional[str] = Header(default=None),
):
    if x_api_key is None or x_client_secret is None:
        raise HTTPException(
            status_code=401,
            detail="Missing authentication credentials. Both X-API-Key and X-Client-Secret headers are required.",
        )

    if not validate_partner_credentials(x_api_key, x_client_secret):
        raise HTTPException(
            status_code=401,
            detail="Invalid authentication credentials. The provided API key or client secret is not valid.",
        )

    return {"api_key": x_api_key, "client_secret": x_client_secret}


@app.post("/payments/process", response_model=PaymentResponse, status_code=200)
async def process_payment(
    payment: PaymentRequest,
    credentials: dict = Depends(verify_partner_credentials),
):
    import uuid

    transaction_id = str(uuid.uuid4())

    return PaymentResponse(
        transaction_id=transaction_id,
        status="success",
        amount=payment.amount,
        currency=payment.currency,
        message=f"Payment of {payment.amount} {payment.currency} processed successfully.",
    )
