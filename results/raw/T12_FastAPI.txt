from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import List, Optional
from datetime import datetime

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    id: int
    username: str
    email: str


class Post(BaseModel):
    id: int
    title: str
    content: str
    published: bool
    author_id: int
    created_at: datetime
    updated_at: datetime


fake_users_db = {
    1: {"id": 1, "username": "john_doe", "email": "john@example.com"},
    2: {"id": 2, "username": "jane_doe", "email": "jane@example.com"},
}

fake_posts_db = [
    {
        "id": 1,
        "title": "My First Draft",
        "content": "This is a draft post.",
        "published": False,
        "author_id": 1,
        "created_at": datetime(2024, 1, 1, 10, 0, 0),
        "updated_at": datetime(2024, 1, 1, 10, 0, 0),
    },
    {
        "id": 2,
        "title": "Published Post",
        "content": "This post is published.",
        "published": True,
        "author_id": 1,
        "created_at": datetime(2024, 1, 2, 10, 0, 0),
        "updated_at": datetime(2024, 1, 2, 10, 0, 0),
    },
    {
        "id": 3,
        "title": "Another Draft",
        "content": "Another unpublished post.",
        "published": False,
        "author_id": 1,
        "created_at": datetime(2024, 1, 3, 10, 0, 0),
        "updated_at": datetime(2024, 1, 3, 10, 0, 0),
    },
    {
        "id": 4,
        "title": "Jane's Draft",
        "content": "This is Jane's draft.",
        "published": False,
        "author_id": 2,
        "created_at": datetime(2024, 1, 4, 10, 0, 0),
        "updated_at": datetime(2024, 1, 4, 10, 0, 0),
    },
]

fake_tokens_db = {
    "token_for_john": 1,
    "token_for_jane": 2,
}


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    user_id = fake_tokens_db.get(token)
    if user_id is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid authentication credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
    user_data = fake_users_db.get(user_id)
    if user_data is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return User(**user_data)


@app.get("/posts/drafts", response_model=List[Post], status_code=status.HTTP_200_OK)
def get_draft_posts(current_user: User = Depends(get_current_user)):
    user_drafts = [
        Post(**post)
        for post in fake_posts_db
        if post["author_id"] == current_user.id and not post["published"]
    ]
    return user_drafts
