import { Controller, ForbiddenException, Get, Param, Req, UnauthorizedException, UseGuards } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { Request } from 'express';
import { WorkspacesService } from './workspaces.service';
import { DocumentsService } from '../documents/documents.service';

type AuthenticatedRequest = Request & {
  user?: {
    id?: string;
    userId?: string;
  };
};

@Controller('workspaces')
export class WorkspacesController {
  constructor(
    private readonly workspacesService: WorkspacesService,
    private readonly documentsService: DocumentsService,
  ) {}

  @Get(':workspaceId/documents')
  @UseGuards(AuthGuard('jwt'))
  async getWorkspaceDocuments(
    @Param('workspaceId') workspaceId: string,
    @Req() req: AuthenticatedRequest,
  ) {
    const userId = req.user?.id ?? req.user?.userId;

    if (!userId) {
      throw new UnauthorizedException();
    }

    const isMember = await this.workspacesService.isMember(userId, workspaceId);

    if (!isMember) {
      throw new ForbiddenException();
    }

    return this.documentsService.findAllByWorkspace(workspaceId);
  }
}
