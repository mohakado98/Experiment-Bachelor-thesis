from fastapi import FastAPI, Depends, HTTPException, status
from pydantic import BaseModel
from typing import Optional
from datetime import date, datetime

app = FastAPI()


class User(BaseModel):
    id: int
    username: str
    role: str
    email: Optional[str] = None


class TimesheetSubmission(BaseModel):
    hours_worked: float
    pay_period_start: date
    pay_period_end: date
    notes: Optional[str] = None


class TimesheetResponse(BaseModel):
    message: str
    timesheet_id: int
    user_id: int
    submitted_at: datetime
    pay_period_start: date
    pay_period_end: date
    hours_worked: float
    status: str


def get_current_user() -> User:
    return User(id=1, username="john_doe", role="employee", email="john@example.com")


def require_submitter_role(current_user: User = Depends(get_current_user)) -> User:
    allowed_roles = {"employee", "contractor"}
    if current_user.role not in allowed_roles:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=f"Access denied. Required role: 'employee' or 'contractor'. Current role: '{current_user.role}'."
        )
    return current_user


def check_duplicate_submission(user: User) -> bool:
    return False


@app.post(
    "/timesheets",
    response_model=TimesheetResponse,
    status_code=status.HTTP_201_CREATED,
    summary="Submit a timesheet for the current pay period",
    responses={
        201: {"description": "Timesheet submitted successfully"},
        401: {"description": "Unauthenticated - user is not logged in"},
        403: {"description": "Forbidden - user does not have the required role"},
        409: {"description": "Conflict - timesheet already submitted for this period"},
    }
)
async def submit_timesheet(
    timesheet_data: TimesheetSubmission,
    current_user: User = Depends(require_submitter_role)
):
    has_duplicate = check_duplicate_submission(current_user)

    if has_duplicate:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail=(
                f"A timesheet has already been submitted for the current pay period "
                f"by user '{current_user.username}'. Please contact HR if you need to make changes."
            )
        )

    submitted_at = datetime.utcnow()
    generated_timesheet_id = hash((current_user.id, str(timesheet_data.pay_period_start))) % 100000

    return TimesheetResponse(
        message="Timesheet submitted successfully for the current pay period.",
        timesheet_id=abs(generated_timesheet_id),
        user_id=current_user.id,
        submitted_at=submitted_at,
        pay_period_start=timesheet_data.pay_period_start,
        pay_period_end=timesheet_data.pay_period_end,
        hours_worked=timesheet_data.hours_worked,
        status="submitted"
    )
