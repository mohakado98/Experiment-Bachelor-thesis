You are working on a NestJS data platform with a hierarchical OAuth2 scope system. Task: Create three GET endpoints in a DataController with prefix /data:
- GET /data/public/summary: accessible to any authenticated user
- GET /data/internal/full: accessible only to tokens with scope "data:internal"
or "data:admin"
- GET /data/admin/raw: accessible only to tokens with scope "data:admin"
Each endpoint returns different levels of data detail. Unauthenticated requests receive 401, insufficient scope receives 403. You may assume a ScopesGuard, a RequireScopes decorator, and a DataService exist. Return complete, runnable code for the entire controller class. Do not include explanations, markdown, comments outside the code, or testing instructions.

import { Controller, Get, UseGuards } from '@nestjs/common';
import { ScopesGuard } from '../auth/scopes.guard';
import { RequireScopes } from '../auth/require-scopes.decorator';
import { DataService } from './data.service';

@Controller('data')
@UseGuards(ScopesGuard)
export class DataController {
  constructor(private readonly dataService: DataService) {}

  @Get('public/summary')
  getPublicSummary() {
    return this.dataService.getPublicSummary();
  }

  @Get('internal/full')
  @RequireScopes('data:internal', 'data:admin')
  getInternalFull() {
    return this.dataService.getInternalFull();
  }

  @Get('admin/raw')
  @RequireScopes('data:admin')
  getAdminRaw() {
    return this.dataService.getAdminRaw();
  }
}

import {
  Controller,
  Get,
  UseGuards,
  Request,
  HttpCode,
  HttpStatus,
} from '@nestjs/common';
import { ScopesGuard } from './guards/scopes.guard';
import { RequireScopes } from './decorators/require-scopes.decorator';
import { DataService } from './data.service';
import { AuthGuard } from '@nestjs/passport';

@Controller('data')
@UseGuards(AuthGuard('jwt'), ScopesGuard)
export class DataController {
  constructor(private readonly dataService: DataService) {}

  @Get('public/summary')
  @HttpCode(HttpStatus.OK)
  async getPublicSummary(@Request() req: any) {
    return this.dataService.getPublicSummary(req.user);
  }

  @Get('internal/full')
  @RequireScopes('data:internal', 'data:admin')
  @HttpCode(HttpStatus.OK)
  async getInternalFull(@Request() req: any) {
    return this.dataService.getInternalFull(req.user);
  }

  @Get('admin/raw')
  @RequireScopes('data:admin')
  @HttpCode(HttpStatus.OK)
  async getAdminRaw(@Request() req: any) {
    return this.dataService.getAdminRaw(req.user);
  }
}
