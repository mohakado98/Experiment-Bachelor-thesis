import {
  Controller,
  Get,
  Param,
  Req,
  UseGuards,
  HttpException,
  HttpStatus,
  ForbiddenException,
  UnauthorizedException,
} from '@nestjs/common';
import { Request } from 'express';
import { ClearanceLevelGuard } from './guards/clearance-level.guard';
import { NeedToKnowGuard } from './guards/need-to-know.guard';
import { AccountSuspensionGuard } from './guards/account-suspension.guard';
import { ClassifiedDocumentsService } from './classified-documents.service';
import { AuthGuard } from '@nestjs/passport';

type ClassificationLevel = 'public' | 'confidential' | 'secret' | 'top_secret';

const VALID_CLASSIFICATION_LEVELS: ClassificationLevel[] = [
  'public',
  'confidential',
  'secret',
  'top_secret',
];

const CLEARANCE_RANK: Record<ClassificationLevel, number> = {
  public: 0,
  confidential: 1,
  secret: 2,
  top_secret: 3,
};

const CLASSIFICATION_ERROR_MESSAGES: Record<ClassificationLevel, string> = {
  public:
    'Access denied: You do not meet the requirements to access public documents.',
  confidential:
    'Access denied: Confidential clearance or higher is required. Verify your clearance level and need-to-know justification.',
  secret:
    'Access denied: Secret clearance or higher is required. This resource is restricted to personnel with verified need-to-know status.',
  top_secret:
    'Access denied: Top Secret clearance is required. Access to this resource is strictly controlled and monitored.',
};

interface AuthenticatedUser {
  id: string;
  clearanceLevel: ClassificationLevel;
  needToKnowCategories: string[];
  isSuspended: boolean;
}

interface AuthenticatedRequest extends Request {
  user: AuthenticatedUser;
}

@Controller('classified')
@UseGuards(AuthGuard('jwt'))
export class ClassifiedController {
  constructor(
    private readonly classifiedDocumentsService: ClassifiedDocumentsService,
  ) {}

  @Get(':classificationLevel/documents')
  @UseGuards(AccountSuspensionGuard, ClearanceLevelGuard, NeedToKnowGuard)
  async getDocumentsByClassificationLevel(
    @Param('classificationLevel') classificationLevel: string,
    @Req() request: AuthenticatedRequest,
  ) {
    if (!request.user) {
      throw new UnauthorizedException(
        'Authentication required: You must be logged in to access classified resources.',
      );
    }

    const normalizedLevel = classificationLevel.toLowerCase() as ClassificationLevel;

    if (!VALID_CLASSIFICATION_LEVELS.includes(normalizedLevel)) {
      throw new HttpException(
        `Invalid classification level: "${classificationLevel}". Valid levels are: ${VALID_CLASSIFICATION_LEVELS.join(', ')}.`,
        HttpStatus.BAD_REQUEST,
      );
    }

    const user = request.user;

    if (!user.clearanceLevel) {
      throw new ForbiddenException(
        'Access denied: No security clearance level found on your account. Contact your security officer.',
      );
    }

    const userClearanceRank = CLEARANCE_RANK[user.clearanceLevel];
    const requiredClearanceRank = CLEARANCE_RANK[normalizedLevel];

    if (userClearanceRank === undefined || requiredClearanceRank === undefined) {
      throw new ForbiddenException(
        'Access denied: Unable to verify clearance levels. Contact your security officer.',
      );
    }

    if (userClearanceRank < requiredClearanceRank) {
      throw new ForbiddenException(
        CLASSIFICATION_ERROR_MESSAGES[normalizedLevel] ??
          'Access denied: Insufficient clearance level.',
      );
    }

    if (user.isSuspended) {
      throw new ForbiddenException(
        'Access denied: Your account access has been suspended pending a security review. Contact your security officer immediately.',
      );
    }

    const hasNeedToKnow =
      await this.classifiedDocumentsService.verifyNeedToKnow(
        user.id,
        normalizedLevel,
      );

    if (!hasNeedToKnow) {
      throw new ForbiddenException(
        `Access denied: You do not have a valid need-to-know justification registered for ${normalizedLevel.replace('_', ' ')} level documents. Submit a need-to-know request to your security officer.`,
      );
    }

    const documents =
      await this.classifiedDocumentsService.getDocumentsByClassificationLevel(
        normalizedLevel,
        user.id,
      );

    return {
      classificationLevel: normalizedLevel,
      accessGranted: true,
      documents,
      accessedBy: user.id,
      accessTimestamp: new Date().toISOString(),
    };
  }
}
