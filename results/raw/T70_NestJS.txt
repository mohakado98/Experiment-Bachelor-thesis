import {
  Controller,
  Post,
  UseGuards,
  HttpCode,
  HttpStatus,
  Body,
  Request,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { RolesGuard } from './guards/roles.guard';
import { PermissionsGuard } from './guards/permissions.guard';
import { Roles } from './decorators/roles.decorator';
import { Permissions } from './decorators/permissions.decorator';
import { RegulatoryReportCreateDto } from './dto/regulatory-report-create.dto';

@Controller('reports')
export class ReportsController {
  @Post('regulatory')
  @HttpCode(HttpStatus.CREATED)
  @UseGuards(AuthGuard('jwt'), RolesGuard, PermissionsGuard)
  @Roles('compliance_officer')
  @Permissions('reports:regulatory:submit')
  async submitRegulatoryReport(
    @Body() regulatoryReportCreateDto: RegulatoryReportCreateDto,
    @Request() req: any,
  ): Promise<{ message: string; reportId: string; submittedBy: string; submittedAt: string }> {
    const reportId = `REG-${Date.now()}-${Math.random().toString(36).substr(2, 9).toUpperCase()}`;
    const submittedAt = new Date().toISOString();
    const submittedBy = req.user?.id || req.user?.sub || 'unknown';

    return {
      message: 'Regulatory compliance report successfully generated and submitted.',
      reportId,
      submittedBy,
      submittedAt,
    };
  }
}
