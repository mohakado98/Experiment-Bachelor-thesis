import { Body, Controller, Delete, Get, Param, Post, Req } from '@nestjs/common';
import type { Request } from 'express';
import { Public } from '../auth/decorators/public.decorator';
import { Permissions } from '../auth/decorators/permissions.decorator';
import { ApiService } from './api.service';

type AuthenticatedRequest = Request & {
  user: unknown;
};

type CreateResourceDto = Record<string, unknown>;

@Controller('api/v1')
export class ApiController {
  constructor(private readonly apiService: ApiService) {}

  @Public()
  @Get('status')
  getStatus() {
    return this.apiService.getStatus();
  }

  @Public()
  @Get('version')
  getVersion() {
    return this.apiService.getVersion();
  }

  @Get('me')
  getMe(@Req() req: AuthenticatedRequest) {
    return req.user;
  }

  @Permissions('resources:write')
  @Post('resources')
  createResource(
    @Body() body: CreateResourceDto,
    @Req() req: AuthenticatedRequest,
  ) {
    return this.apiService.createResource(body, req.user);
  }

  @Permissions('resources:delete')
  @Delete('resources/:id')
  deleteResource(
    @Param('id') id: string,
    @Req() req: AuthenticatedRequest,
  ) {
    return this.apiService.deleteResource(id, req.user);
  }
}

import {
  Controller,
  Get,
  Post,
  Delete,
  Param,
  Body,
  HttpCode,
  HttpStatus,
} from '@nestjs/common';
import { SetMetadata } from '@nestjs/common';

export const IS_PUBLIC_KEY = 'isPublic';
export const Public = () => SetMetadata(IS_PUBLIC_KEY, true);

export const PERMISSIONS_KEY = 'permissions';
export const RequirePermissions = (...permissions: string[]) =>
  SetMetadata(PERMISSIONS_KEY, permissions);

@Controller('api/v1')
export class ApiController {
  @Public()
  @Get('status')
  @HttpCode(HttpStatus.OK)
  getStatus() {
    return {
      status: 'ok',
      timestamp: new Date().toISOString(),
      uptime: process.uptime(),
    };
  }

  @Public()
  @Get('version')
  @HttpCode(HttpStatus.OK)
  getVersion() {
    return {
      version: process.env.APP_VERSION ?? '1.0.0',
      apiVersion: 'v1',
      nodeVersion: process.version,
    };
  }

  @Get('me')
  @HttpCode(HttpStatus.OK)
  getMe() {
    return {
      message: 'Authenticated user profile',
    };
  }

  @Post('resources')
  @RequirePermissions('resources:write')
  @HttpCode(HttpStatus.CREATED)
  createResource(@Body() body: Record<string, unknown>) {
    return {
      message: 'Resource created successfully',
      data: body,
      createdAt: new Date().toISOString(),
    };
  }

  @Delete('resources/:id')
  @RequirePermissions('resources:delete')
  @HttpCode(HttpStatus.OK)
  deleteResource(@Param('id') id: string) {
    return {
      message: `Resource with id ${id} deleted successfully`,
      deletedAt: new Date().toISOString(),
    };
  }
}
