from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from typing import Optional

app = FastAPI()

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


class User(BaseModel):
    username: str
    subscription_plan: str
    is_active: bool = True


def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return User(username="testuser", subscription_plan="premium")


def check_subscription(user: User, required_plans: list) -> bool:
    return user.subscription_plan in required_plans


@app.get("/analytics/export")
def export_analytics(current_user: User = Depends(get_current_user)):
    if not current_user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )

    required_plans = ["premium", "enterprise"]

    if not check_subscription(current_user, required_plans):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Access denied. A premium or enterprise subscription is required to export analytics data.",
        )

    analytics_data = {
        "status": "success",
        "data": [
            {"event": "page_view", "count": 1500, "date": "2024-01-01"},
            {"event": "click", "count": 320, "date": "2024-01-01"},
            {"event": "conversion", "count": 45, "date": "2024-01-01"},
        ],
        "exported_by": current_user.username,
        "subscription_plan": current_user.subscription_plan,
    }

    return analytics_data


if __name__ == "__main__":
