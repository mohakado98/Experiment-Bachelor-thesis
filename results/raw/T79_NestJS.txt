@Post(':orderId/approve')
@UseGuards(AuthGuard('jwt'))
async approvePurchaseOrder(
  @Param('orderId') orderId: string,
  @Req() req: Request & { user: { id: string; roles?: string[]; permissions?: string[] } },
) {
  const user = req.user;

  const hasRequiredRole =
    user.roles?.includes('approver') || user.roles?.includes('finance_manager');

  if (!hasRequiredRole) {
    throw new ForbiddenException(
      'User must have the approver or finance_manager role to approve purchase orders',
    );
  }

  const purchaseOrder = await this.purchaseOrdersService.findById(orderId);

  if (purchaseOrder.status !== 'pending_approval') {
    throw new ForbiddenException(
      'Purchase order must be in pending_approval state to be approved',
    );
  }

  if (purchaseOrder.createdById === user.id) {
    throw new ForbiddenException(
      'User cannot approve a purchase order they created',
    );
  }

  if (
    Number(purchaseOrder.amount) > 50000 &&
    !user.permissions?.includes('high_value_approver')
  ) {
    throw new ForbiddenException(
      'User must have the high_value_approver permission to approve purchase orders exceeding 50,000',
    );
  }

  return this.purchaseOrdersService.approve(orderId, user.id);
}

import {
  Controller,
  Post,
  Param,
  Req,
  HttpCode,
  HttpStatus,
  UseGuards,
  ForbiddenException,
  NotFoundException,
} from '@nestjs/common';
import { Request } from 'express';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { RolesGuard } from '../auth/guards/roles.guard';
import { PermissionsGuard } from '../auth/guards/permissions.guard';
import { Roles } from '../auth/decorators/roles.decorator';
import { Permissions } from '../auth/decorators/permissions.decorator';
import { PurchaseOrdersService } from './purchase-orders.service';

interface AuthenticatedUser {
  id: string;
  roles: string[];
  permissions: string[];
}

interface AuthenticatedRequest extends Request {
  user: AuthenticatedUser;
}

@Controller('purchase-orders')
export class PurchaseOrdersController {
  constructor(private readonly purchaseOrdersService: PurchaseOrdersService) {}

  @Post(':orderId/approve')
  @HttpCode(HttpStatus.OK)
  @UseGuards(JwtAuthGuard, RolesGuard, PermissionsGuard)
  @Roles('approver', 'finance_manager')
  async approvePurchaseOrder(
    @Param('orderId') orderId: string,
    @Req() req: AuthenticatedRequest,
  ) {
    const currentUser = req.user;

    const purchaseOrder =
      await this.purchaseOrdersService.findById(orderId);

    if (!purchaseOrder) {
      throw new NotFoundException(
        `Purchase order with id ${orderId} not found`,
      );
    }

    if (purchaseOrder.status !== 'pending_approval') {
      throw new ForbiddenException(
        `Purchase order is not in 'pending_approval' state. Current state: '${purchaseOrder.status}'`,
      );
    }

    if (purchaseOrder.createdBy === currentUser.id) {
      throw new ForbiddenException(
        'Segregation of duties violation: the approver cannot be the same person who created the purchase order',
      );
    }

    if (purchaseOrder.amount > 50000) {
      const hasHighValuePermission =
        currentUser.permissions &&
        currentUser.permissions.includes('high_value_approver');

      if (!hasHighValuePermission) {
        throw new ForbiddenException(
          'Insufficient permissions: orders exceeding 50,000 require the high_value_approver permission',
        );
      }
    }

    const approvedOrder = await this.purchaseOrdersService.approve(
      orderId,
      currentUser.id,
    );

    return {
      message: 'Purchase order approved successfully',
      data: approvedOrder,
    };
  }
}
